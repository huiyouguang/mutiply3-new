async function readHealthData(date) {
  const healthPath = `TaskNotes/健康数据/${date}.md`;
  const file = app.vault.getAbstractFileByPath(healthPath);
  if (!file) return null;
  
  const content = await app.vault.read(file);
  try {
    const frontmatterMatch = content.match(/^---\n([\s\S]*?)\n---/);
    if (frontmatterMatch) {
      const yaml = frontmatterMatch[1];
      const data = {};
      yaml.split('\n').forEach(line => {
        const [key, ...valueParts] = line.split(':');
        if (key && valueParts.length > 0) {
          let value = valueParts.join(':').trim();
          if (value.startsWith('[') || value.startsWith('{')) {
            try { value = JSON.parse(value); } catch(e) {}
          } else if (!isNaN(value) && value !== '') {
            value = parseFloat(value);
          }
          data[key.trim()] = value;
        }
      });
      return data;
    }
  } catch(e) { console.error('Parse health file error:', e); }
  
  return null;
}

// 获取文件夹中的文件和子文件夹
function getStudyFiles(folderPath) {
  const directFiles = allFilePaths
    .filter(p => p.startsWith(folderPath + '/') && p.endsWith('.md'))
    .map(p => ({
      type: 'file',
      name: p.replace(folderPath + '/', '').replace('.md', ''),
      fullPath: p
    }));
  
  // 获取直接子文件夹
  const subFolders = new Set();
  directFiles.forEach(f => {
    const parts = f.name.split('/');
    if (parts.length > 1) {
      subFolders.add(parts[0]);
    }
  });
  
  const files = directFiles
    .filter(f => !f.name.includes('/')) // 只保留直接文件
    .slice(0, 15);
  
  const folders = Array.from(subFolders).slice(0, 10).map(name => ({
    type: 'folder',
    name: name,
    fullPath: `${folderPath}/${name}`
  }));
  
  return { files, folders };
}

// 打开文件列表（打开侧边栏并定位到文件夹）
function openAllStudyFiles(folderPath) {
  // 先展开侧边栏（如果收起来的话）
  const workspace = app.workspace;
  
  // 尝试展开左侧边栏
  const leftLeaf = workspace.leftSplit;
  if (leftLeaf && !leftLeaf.containerEl.classList.contains('is-loaded')) {
    // 如果侧边栏还没加载，尝试打开文件列表
    workspace.getLeftLeaf(false).then((leaf) => {
      leaf.openFile(app.vault.getAbstractFileByPath('04成长/知识管理'));
    });
  } else if (leftLeaf) {
    // 确保侧边栏可见
    leftLeaf.containerEl.style.display = '';
  }
  
  // 尝试打开目标文件夹
  const targetFile = app.vault.getAbstractFileByPath(folderPath);
  if (targetFile) {
    // 如果是文件夹，在左侧边栏打开
    workspace.getLeftLeaf(false).then((leaf) => {
      leaf.openFile(targetFile);
    });
  } else {
    // 如果文件夹不存在，尝试打开该路径下的任意文件
    const { files } = getStudyFiles(folderPath);
    if (files.length > 0) {
      const firstFile = app.vault.getAbstractFileByPath(files[0].fullPath);
      if (firstFile) {
        workspace.getLeftLeaf(false).then((leaf) => {
          leaf.openFile(firstFile);
        });
      }
    }
  }
  
  // 收起文件列表
  setTimeout(() => {
    const listArea = document.getElementById('study-file-list');
    if (listArea) {
      studyListExpanded = false;
      currentStudyPath = null;
      listArea.style.display = 'none';
    }
  }, 300);
}

function flattenChildren(node) {
  let flat = [];
  node.childrenTasks.forEach(child => {
    flat.push(child);

}

function saveThinoNote(content) {
  if (!content.trim()) return;
  const timeStr = `${String(NOW.getHours()).padStart(2, '0')}:${String(NOW.getMinutes()).padStart(2, '0')}`;
  const noteLine = `- ${timeStr} ${content.trim()}`;
  
  (async () => {
    const existing = app.vault.getAbstractFileByPath(todayNotePath);
    if (existing) {
      const current = await app.vault.read(existing);
      if (current.includes('✏随笔感悟')) {
        const updated = current.replace(/✏随笔感悟\n/, `✏随笔感悟\n${noteLine}\n`);
        await app.vault.modify(existing, updated);
      } else {
        await app.vault.modify(existing, current + `\n\n## ✏随笔感悟\n${noteLine}\n`);
      }
    } else {
      const newContent = `---\ncreated: ${NOW.toISOString().slice(0, 10)}\n---\n\n## ✏随笔感悟\n${noteLine}\n`;
      await app.vault.create(todayNotePath, newContent);
    }
    const statusEl = document.getElementById('thino-status');
    if (statusEl) {
      statusEl.innerHTML = `✅ 已保存到 ${todayNotePath}`;
      statusEl.style.color = C.green;
      setTimeout(() => { statusEl.innerHTML = ''; }, 2000);
    }
    document.getElementById('thino-input').value = '';
  })();
}
