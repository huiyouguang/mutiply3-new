async function readHealthFile(date) {
  const filePath = getHealthFilePath(date);
  const file = app.vault.getAbstractFileByPath(filePath);
  
  if (!file) return null;
  
  const content = await app.vault.read(file);
  
  try {
    const frontmatterMatch = content.match(/^---\n([\s\S]*?)\n---/);
    if (frontmatterMatch) {
      const yaml = frontmatterMatch[1];
      const data = {};
      yaml.split('\n').forEach(line => {
        const [key, ...valueParts] = line.split(':');
        if (key && valueParts.length > 0) {
          let value = valueParts.join(':').trim();
          if (value.startsWith('[') || value.startsWith('{')) {
            try { value = JSON.parse(value); } catch(e) {}
          } else if (!isNaN(value) && value !== '') {
            value = parseFloat(value);
          }
          data[key.trim()] = value;
        }
      });
      return data;
    }
  } catch(e) { console.error('Parse health file error:', e); }
  
  return null;
}

    ctx.fill();
    
    ctx.beginPath();
    ctx.arc(x, y, 5, 0, Math.PI * 2);
    ctx.strokeStyle = hexToRgba(C.accent, 0.4);
    ctx.lineWidth = 2;
    ctx.stroke();
  });

async function saveWeight() {
  const weight = parseFloat(document.getElementById('weight-value').value);
  if (!weight) return;
  const date = document.getElementById('weight-date').value || today;
  
  const existingData = await readHealthFile(date) || {};
  existingData.weight = weight;
  
  await writeHealthFile(date, existingData);
  await renderWeight();
}

async function deleteWeight(date) {
  const existingData = await readHealthFile(date) || {};
  delete existingData.weight;
  
  if (Object.keys(existingData).length > 1) {
    await writeHealthFile(date, existingData);
  } else {
    const file = app.vault.getAbstractFileByPath(getHealthFilePath(date));
    if (file) await app.vault.delete(file);
  }
  
  await renderWeight();
}

async function renderWeight() {
  const allData = await loadAllHealthData();
  const weightData = {};
  
  Object.entries(allData).forEach(([date, data]) => {
    if (data.weight) weightData[date] = data.weight;
  });
  
  const sortedDates = Object.keys(weightData).sort();
  const recentData = sortedDates.slice(-10);
  
  const latestWeight = sortedDates.length > 0 ? weightData[sortedDates[sortedDates.length-1]] : null;
  const firstWeight = sortedDates.length > 0 ? weightData[sortedDates[0]] : null;
  const diff = latestWeight && firstWeight ? (latestWeight - firstWeight).toFixed(1) : null;
  
  let chartHtml = '';
  if (recentData.length > 0) {
    const maxWeight = Math.max(...recentData.map(d => weightData[d]));
    const minWeight = Math.min(...recentData.map(d => weightData[d]));
    const range = maxWeight - minWeight || 1;
    
    chartHtml = `<div style="display:flex;align-items:flex-end;justify-content:space-between;height:80px;padding:10px 0;">${recentData.map(date => {
      const w = weightData[date];
      const height = ((w - minWeight) / range) * 60 + 20;
      return `<div style="display:flex;flex-direction:column;align-items:center;gap:4px;"><div style="width:20px;height:${height}px;background:${C.accent};border-radius:4px 4px 0 0;"></div><span style="font-size:9px;color:${C.sub};">${date.slice(5)}</span></div>`;
    }).join('')}</div>`;
  }
  
  setHealthModuleContent('weightTracker', `
    <div style="display:flex;gap:16px;">
      <div style="flex:1;">
        <div style="display:flex;justify-content:space-between;align-items:center;margin-bottom:12px;">
          <div style="font-size:13px;font-weight:600;color:${C.text};">体重趋势</div>
          <div style="font-size:12px;color:${C.sub};">${diff !== null ? (diff > 0 ? '↑+' : '↓') + diff + 'kg' : ''}</div>
        </div>
        ${chartHtml || '<div style="text-align:center;padding:30px;color:'+C.sub+';font-size:12px;">暂无体重记录</div>'}
        <div style="margin-top:12px;">${recentData.map(date => `<div style="display:flex;justify-content:space-between;align-items:center;padding:6px 0;border-bottom:1px dashed ${C.border};"><span style="font-size:12px;color:${C.sub};">${date}</span><div style="display:flex;align-items:center;gap:12px;"><span style="font-size:14px;font-weight:600;color:${C.text};">${weightData[date]}kg</span><button class="health-btn" onclick="deleteWeight('${date}')" style="font-size:10px;">删除</button></div></div>`).join('')}</div>
      </div>
      <div style="width:200px;background:${C.card2};border-radius:12px;padding:14px;">
        <div style="font-size:13px;font-weight:600;color:${C.text};margin-bottom:12px;">记录体重</div>
        <div style="display:flex;flex-direction:column;gap:10px;">
          <div>
            <label style="font-size:11px;color:${C.sub};display:block;margin-bottom:4px;">日期</label>
            <input type="date" id="weight-date" value="${today}" style="width:100%;padding:8px;border:1px solid ${C.border};border-radius:8px;background:${C.card};color:${C.text};font-size:12px;">
          </div>
          <div>
            <label style="font-size:11px;color:${C.sub};display:block;margin-bottom:4px;">体重（kg）</label>
            <input type="number" id="weight-value" step="0.1" placeholder="65.5" style="width:100%;padding:8px;border:1px solid ${C.border};border-radius:8px;background:${C.card};color:${C.text};font-size:12px;">

async function renderHealthCalendar() {
  const allData = await loadAllHealthData();
  const year = NOW.getFullYear();
  const month = NOW.getMonth();
  const firstDay = new Date(year, month, 1);
  const lastDay = new Date(year, month + 1, 0);
  const daysInMonth = lastDay.getDate();
  const startDay = firstDay.getDay();
  
  let calendarHtml = '<div class="health-grid-7" style="text-align:center;font-size:11px;color:'+C.sub+';margin-bottom:8px;">';
  ['日','一','二','三','四','五','六'].forEach(d => { calendarHtml += '<div>'+d+'</div>'; });
  calendarHtml += '</div>';
  
  calendarHtml += '<div class="health-grid-7" style="text-align:center;">';
  
  for (let i = 0; i < startDay; i++) { calendarHtml += '<div></div>'; }
  
  for (let d = 1; d <= daysInMonth; d++) {
    const dateStr = year+'-'+String(month+1).padStart(2,'0')+'-'+String(d).padStart(2,'0');
    const dayData = allData[dateStr] || {};
    const hasSleep = !!dayData.sleepDuration;
    const hasWater = (dayData.waterCount || 0) >= 8;
    const hasExercise = (dayData.exercises || []).length > 0;
    const hasMood = !!dayData.mood;
    
    const completed = [hasSleep, hasWater, hasExercise, hasMood].filter(Boolean).length;
    const intensity = completed === 4 ? 4 : completed === 3 ? 3 : completed === 2 ? 2 : completed === 1 ? 1 : 0;
    
    const heatColors = [C.card2, C.heat1, C.heat2, C.heat3, C.heat4];
    
    calendarHtml += `<div style="width:28px;height:28px;border-radius:50%;background:${heatColors[intensity]};display:flex;align-items:center;justify-content:center;font-size:12px;color:${intensity >= 3 ? C.text : C.sub};margin:auto;cursor:pointer;" title="${completed}/4 项健康记录">${d}</div>`;
  }
  
  calendarHtml += '</div>';
  
  setHealthModuleContent('healthCalendar', `
    <div>
      <div style="display:flex;justify-content:space-between;align-items:center;margin-bottom:16px;">
        <div style="font-size:14px;font-weight:600;color:${C.text};">${year}年${month+1}月 · 健康打卡</div>
        <div style="display:flex;align-items:center;gap:8px;">
          <span style="font-size:10px;color:${C.sub};">少</span>
          ${[C.card2, C.heat1, C.heat2, C.heat3, C.heat4].map(c => '<div style="width:12px;height:12px;border-radius:2px;background:'+c+'"></div>').join('')}
          <span style="font-size:10px;color:${C.sub};">多</span>
        </div>
      </div>
      ${calendarHtml}
    </div>
  `);
}

const GOALS_FILE_PATH = '04 TaskNotes/健康数据/健康目标.md';

];

async function readGoalsFile() {
  await ensureHealthFolder();
  const file = app.vault.getAbstractFileByPath(GOALS_FILE_PATH);
  
  if (!file) return defaultGoals;
  
  const content = await app.vault.read(file);
  try {
    const frontmatterMatch = content.match(/^---\n([\s\S]*?)\n---/);
    if (frontmatterMatch) {
      const yaml = frontmatterMatch[1];
      const goalsLine = yaml.split('\n').find(l => l.startsWith('goals:'));
      if (goalsLine) {
        const goalsStr = goalsLine.replace('goals:', '').trim();
        return JSON.parse(goalsStr);
      }
    }
  } catch(e) { console.error('Parse goals error:', e); }
  
  return defaultGoals;
}

async function writeGoalsFile(goals) {
  await ensureHealthFolder();
  const file = app.vault.getAbstractFileByPath(GOALS_FILE_PATH);
  
  const fileContent = `---
goals: ${JSON.stringify(goals)}
---

# 🎯 健康目标

## 目标列表

${goals.map(g => `- ${g.icon} ${g.name}: ${g.current}/${g.target}${g.unit} (${Math.min(100, (g.current / g.target) * 100).toFixed(0)}%)`).join('\n')}
`;

  if (file) {
    await app.vault.modify(file, fileContent);
  } else {
    await app.vault.create(GOALS_FILE_PATH, fileContent);
  }
}

async function updateGoal(id, value) {
  const goals = await readGoalsFile();
  const goal = goals.find(g => g.id === id);
  if (goal) {
    goal.current = parseFloat(value);
    await writeGoalsFile(goals);
    await renderHealthGoals();
  }
}

async function addGoal() {
  const name = prompt('目标名称：');
  if (!name) return;
  const target = parseFloat(prompt('目标值：'));
  if (!target) return;
  const unit = prompt('单位：') || '';
  const icon = prompt('图标：') || '🎯';
  
  const goals = await readGoalsFile();
  goals.push({ id: Date.now(), name, target, current: 0, unit, icon });
  await writeGoalsFile(goals);
  await renderHealthGoals();
}

async function deleteGoal(id) {
  if (confirm('确定删除这个目标吗？')) {
    const goals = await readGoalsFile();
    const filteredGoals = goals.filter(g => g.id !== id);
    await writeGoalsFile(filteredGoals);
    await renderHealthGoals();
  }
}
