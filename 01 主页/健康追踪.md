function renderHealthCalendar() {
  const year = NOW.getFullYear();
  const month = NOW.getMonth();
  const firstDay = new Date(year, month, 1);
  const lastDay = new Date(year, month + 1, 0);
  const daysInMonth = lastDay.getDate();
  const startDay = firstDay.getDay();
  
  let calendarHtml = '<div class="health-grid-7" style="text-align:center;font-size:11px;color:'+C.sub+';margin-bottom:8px;">';
  ['日','一','二','三','四','五','六'].forEach(d => { calendarHtml += '<div>'+d+'</div>'; });
  calendarHtml += '</div>';
  
  calendarHtml += '<div class="health-grid-7" style="text-align:center;">';
  
  for (let i = 0; i < startDay; i++) { calendarHtml += '<div></div>'; }
  
  for (let d = 1; d <= daysInMonth; d++) {
    const dateStr = year+'-'+String(month+1).padStart(2,'0')+'-'+String(d).padStart(2,'0');
    const hasSleep = !!sleepData[dateStr];
    const hasWater = waterData[dateStr] >= waterGoal;
    const hasExercise = exerciseData[dateStr] && exerciseData[dateStr].length > 0;
    const hasMood = !!moodData[dateStr];
    
    const completed = [hasSleep, hasWater, hasExercise, hasMood].filter(Boolean).length;
    const intensity = completed === 4 ? 4 : completed === 3 ? 3 : completed === 2 ? 2 : completed === 1 ? 1 : 0;
    
    const heatColors = [C.card2, C.heat1, C.heat2, C.heat3, C.heat4];
    
    calendarHtml += `<div style="width:28px;height:28px;border-radius:50%;background:${heatColors[intensity]};display:flex;align-items:center;justify-content:center;font-size:12px;color:${intensity >= 3 ? C.text : C.sub};margin:auto;cursor:pointer;" title="${completed}/4 项健康记录">${d}</div>`;
  }
  
  calendarHtml += '</div>';
  
  setHealthModuleContent('healthCalendar', `
    <div>
      <div style="display:flex;justify-content:space-between;align-items:center;margin-bottom:16px;">
        <div style="font-size:14px;font-weight:600;color:${C.text};">${year}年${month+1}月 · 健康打卡</div>
        <div style="display:flex;align-items:center;gap:8px;">
          <span style="font-size:10px;color:${C.sub};">少</span>
          ${[C.card2, C.heat1, C.heat2, C.heat3, C.heat4].map(c => '<div style="width:12px;height:12px;border-radius:2px;background:'+c+'"></div>').join('')}
          <span style="font-size:10px;color:${C.sub};">多</span>
        </div>
      </div>
      ${calendarHtml}
    </div>
  `);
}

let healthGoals = JSON.parse(localStorage.getItem('health_goals') || JSON.stringify([
  { id: 1, name: '每月运动 20 天', target: 20, current: 0, unit: '天', icon: '🏃' },
  { id: 2, name: '体重降至 65kg', target: 65, current: 70, unit: 'kg', icon: '⚖️' },
  { id: 3, name: '每月阅读 4 本书', target: 4, current: 0, unit: '本', icon: '📚' },
  { id: 4, name: '每天睡够 7 小时', target: 7, current: 6.5, unit: '小时', icon: '😴' },
]));

function updateGoal(id, value) {
  const goal = healthGoals.find(g => g.id === id);
  if (goal) {
    goal.current = parseFloat(value);
    localStorage.setItem('health_goals', JSON.stringify(healthGoals));
    renderHealthGoals();
  }
}

function addGoal() {
  const name = prompt('目标名称：');
  if (!name) return;
  const target = parseFloat(prompt('目标值：'));
  if (!target) return;
  const unit = prompt('单位：') || '';
  const icon = prompt('图标：') || '🎯';
  healthGoals.push({ id: Date.now(), name, target, current: 0, unit, icon });
  localStorage.setItem('health_goals', JSON.stringify(healthGoals));
  renderHealthGoals();
}

function deleteGoal(id) {
  if (confirm('确定删除这个目标吗？')) {
    healthGoals = healthGoals.filter(g => g.id !== id);
    localStorage.setItem('health_goals', JSON.stringify(healthGoals));
    renderHealthGoals();
  }
}

function renderHealthGoals() {
  const goalsHtml = healthGoals.map(g => {
    const pct = Math.min(100, (g.current / g.target) * 100);
    return `
      <div style="padding:12px;background:${C.card2};border-radius:10px;margin-bottom:10px;">
        <div style="display:flex;justify-content:space-between;align-items:center;margin-bottom:8px;">
          <div style="display:flex;align-items:center;gap:8px;">
            <span style="font-size:16px;">${g.icon}</span>
            <span style="font-size:13px;font-weight:600;color:${C.text};">${g.name}</span>
          </div>
          <div style="display:flex;align-items:center;gap:8px;">
            <span style="font-size:14px;font-weight:700;color:${C.accent};">${fmt(g.current)}${g.unit}</span>
            <span style="font-size:11px;color:${C.sub};">/ ${g.target}${g.unit}</span>
            <button onclick="deleteGoal(${g.id})" style="border:none;background:none;color:${C.sub};cursor:pointer;">🗑️</button>
          </div>
        </div>
        <div style="height:6px;background:${C.border};border-radius:3px;overflow:hidden;">

    <div>

        <div style="font-size:13px;font-weight:600;color:${C.text};">健康目标</div>
        <button class="health-btn" onclick="addGoal()" style="font-size:11px;">➕ 添加</button>
      </div>
      ${goalsHtml}
    </div>
  `);
}
