function toggleTask(path, line) {
  const file = app.vault.getAbstractFileByPath(path);
  if (!file) return;
  app.vault.read(file).then(content => {
    const lines = content.split('\n');
    const taskLine = lines[line];
    if (!taskLine) return;
    let newLine;
    if (taskLine.includes('[ ]')) newLine = taskLine.replace('[ ]', '[x]');
    else if (taskLine.includes('[x]')) newLine = taskLine.replace('[x]', '[ ]');
    else if (taskLine.includes('[/]')) newLine = taskLine.replace('[/]', '[x]');
    else if (taskLine.includes('[>]')) newLine = taskLine.replace('[>]', '[x]');
    else return;
    if (newLine !== taskLine) {
      lines[line] = newLine;
      app.vault.modify(file, lines.join('\n'));
    }
  });
}
